Connection

DataBridge.connect()

Establishes a connection to a database.

Syntax

DataBridge.connect(config: ConnectionConfig): Promise

Parameters

interface ConnectionConfig {
    provider: "postgres" | "mysql" | "mongodb" | "sqlite";
    url?: string;
    host?: string;
    port?: number;
    user?: string;
    password?: string;
    database?: string;
    options?: Record;
}

Example

const db = await DataBridge.connect({
    provider: "postgres",
    url: "postgres://user:password@localhost:5432/database"
});

db.model()

Creates a new model for a collection or table.

Syntax

await db.model(name: string, schema: Schema): Model

Example

const User = await db.model("User", new Schema({
    name: String,
    email: String
}));
