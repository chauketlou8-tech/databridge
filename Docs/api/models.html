Models

Model.create()

Creates a new record.

Syntax

model.create(data: Record): Promise

Example

await User.create({
    name: "John",
    email: "john@example.com",
    age: 21
});

Model.find()

Finds all records matching a query.

Syntax

model.find(query?: Query): Promise

Example

const users = await User.find({ age: { $gt: 18 } });

Model.findOne()

Finds a single record matching a query.

Syntax

model.findOne(query: Query): Promise

Example

const user = await User.findOne({ email: "john@example.com" });

Model.update()

Updates records matching a query.

Syntax

model.update(filter: Query, data: Record): Promise

Example

await User.update(
    { name: "John" },
    { age: 22 }
);

Model.delete()

Deletes records matching a query.

Syntax

model.delete(query: Query): Promise

Example

await User.delete({ name: "John" });
