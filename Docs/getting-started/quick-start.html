Quick Start

Get up and running with DataBridge in 30 seconds.

Connect to a Database

import { DataBridge, Schema } from "databridge";

const db = await DataBridge.connect({
    provider: "postgres",
    url: "postgres://localhost:5432/myapp"
});

Define a Model

const User = db.model("User", new Schema({
    name: String,
    email: String,
    age: Number
}));

Create a Record

await User.create({
    name: "John",
    email: "john@example.com",
    age: 21
});

Query Records

const users = await User.find();
console.log(users);

Complete Example

import { DataBridge, Schema } from "databridge";

async function main() {
    const db = await DataBridge.connect({
        provider: "postgres",
        url: "postgres://localhost:5432/myapp"
    });

    const User = db.model("User", new Schema({
        name: String,
        email: String,
        age: Number
    }));

    await User.create({
        name: "John",
        email: "john@example.com",
        age: 21
    });

    const users = await User.find();
    console.log(users);
}

main();
