MySQL Guide

Connection

const db = await DataBridge.connect({
    provider: "mysql",
    host: "localhost",
    user: "root",
    password: "password",
    database: "myapp"
});

Connection Options

const db = await DataBridge.connect({
    provider: "mysql",
    host: "localhost",
    port: 3306,
    user: "root",
    password: "password",
    database: "myapp",
    options: {
        maxConnections: 10,
        idleTimeout: 30000,
        ssl: true
    }
});

Example

import { DataBridge, Schema } from "databridge";

const db = await DataBridge.connect({
    provider: "mysql",
    host: "localhost",
    user: "root",
    password: "password",
    database: "myapp"
});

const Product = db.model("Product", new Schema({
    name: String,
    price: Number,
    category: String
}));

await Product.create({
    name: "Laptop",
    price: 1200,
    category: "Technology"
});

const products = await Product.find();
