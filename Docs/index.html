Quick Example

Get started in 30 seconds with DataBridge.

import { DataBridge, Schema } from "databridge";

// Connect to your database
const db = await DataBridge.connect({
    provider: "postgres",
    url: "postgres://localhost:5432/myapp"
});

// Define your model
const User = db.model("User", new Schema({
    name: String,
    email: String,
    age: Number
}));

// Create a record
await User.create({
    name: "John",
    email: "john@example.com",
    age: 21
});

// Query records
const users = await User.find();
console.log(users);
