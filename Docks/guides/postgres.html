PostgreSQL Guide

Connection

const db = await DataBridge.connect({
    provider: "postgres",
    url: "postgres://user:password@localhost:5432/database"
});

Connection Options

const db = await DataBridge.connect({
    provider: "postgres",
    host: "localhost",
    port: 5432,
    user: "postgres",
    password: "password",
    database: "myapp",
    options: {
        maxConnections: 10,
        idleTimeout: 30000,
        ssl: true
    }
});

Example

import { DataBridge, Schema } from "databridge";

const db = await DataBridge.connect({
    provider: "postgres",
    url: "postgres://localhost:5432/myapp"
});

const User = db.model("User", new Schema({
    name: String,
    email: String,
    age: Number
}));

await User.create({
    name: "John",
    email: "john@example.com",
    age: 21
});

const users = await User.find();
