SQLite Guide

Connection

const db = await DataBridge.connect({
    provider: "sqlite",
    filename: "./database.sqlite"
});

Connection Options

const db = await DataBridge.connect({
    provider: "sqlite",
    filename: "./database.sqlite",
    options: {
        readonly: false,
        timeout: 5000
    }
});

In-Memory Database

const db = await DataBridge.connect({
    provider: "sqlite",
    filename: ":memory:"
});

Example

import { DataBridge, Schema } from "databridge";

const db = await DataBridge.connect({
    provider: "sqlite",
    filename: "./myapp.sqlite"
});

const Task = db.model("Task", new Schema({
    title: String,
    description: String,
    completed: Boolean
}));

await Task.create({
    title: "Write documentation",
    description: "Complete the user guide",
    completed: false
});

const tasks = await Task.find();
