MongoDB Guide

Connection

const db = await DataBridge.connect({
    provider: "mongodb",
    url: "mongodb://localhost:27017/database"
});

Connection Options

const db = await DataBridge.connect({
    provider: "mongodb",
    url: "mongodb://localhost:27017/database",
    options: {
        useNewUrlParser: true,
        useUnifiedTopology: true,
        maxPoolSize: 10,
        minPoolSize: 2
    }
});

Example

import { DataBridge, Schema } from "databridge";

const db = await DataBridge.connect({
    provider: "mongodb",
    url: "mongodb://localhost:27017/myapp"
});

const Post = db.model("Post", new Schema({
    title: String,
    content: String,
    author: String,
    tags: [String]
}));

await Post.create({
    title: "Hello World",
    content: "This is my first post",
    author: "John",
    tags: ["intro", "hello"]
});

const posts = await Post.find();
